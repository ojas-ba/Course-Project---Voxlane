================================================================================
PIPECAT VOICE AGENT PIPELINE - COMPILING FOR EXTERNAL REVIEW
================================================================================

This document aggregates the entire code and architectural context of the local-development
voice agent pipeline. It is structured to help an external AI agent or human reviewer
understand, audit, and evaluate the code quality, design decisions, and framework integrations.

--------------------------------------------------------------------------------
PROJECT & FRAMEWORK SPECS
--------------------------------------------------------------------------------
1. Python Version: 3.11+ (running 3.12.1 in local venv).
2. Framework: FastAPI (ASGI router) + Pipecat (v1.2.1) + Motor (async MongoDB client).
3. Providers: Deepgram (STT, using model="flux" for low turn latency), Cerebras (LLM), 
   and Cartesia (TTS, using model="sonic-3.5" in SENTENCE-aggregation mode with 0ms server delay).
4. Audio & Telephony Contract: Exotel dynamic WebSocket voicebot applet. All transport 
   and pipeline sample rates are hardcoded to 16000Hz (16kHz 16-bit Mono PCM).
5. Architecture Pattern: Spec-first, phase-gated design. Clean boundary separation between 
   telephony routes ( FastAPIO ), pipeline execution ( Pipecat ), and state/metrics management ( Pydantic/MongoDB ).

--------------------------------------------------------------------------------
ARCHITECTURE FLOW & STATE LIFECYCLE
--------------------------------------------------------------------------------
- [Outbound Dialing] -> Triggered via browser dashboard -> REST API call to Exotel.
- [Answer Callback]  -> Exotel sends GET to /telephony/webhook -> Returns dynamic WSS URL.
- [WebSocket Setup]  -> Exotel connects to WS /telephony/exotel -> Handshake accepted.
- [Session Bootstrap] -> Session created in memory (ACTIVE_SESSIONS) with immutable config snapshot.
- [Transport Setup]  -> Exotel transport constructed with ExotelFrameSerializer.
- [Pipeline Run]     -> Deepgram + Cerebras + Cartesia graph assembled, observers registered, and runner started.
- [Observing Timings]-> UserBotLatencyObserver captures firstbot latency and turn-by-turn event breakdowns.
- [Client Hangup]    -> Transport detects disconnect -> Cancels task -> Unwinds runner -> Router finalizes metrics -> Call log persisted to MongoDB -> Session deleted from registry.

FILE 1: app/runtime.py
================================================================================
- Context: Single source of truth for runtime session registry and state helpers.
- Key Design Decisions:
  * Uses simple dictionaries with typed schemas (TypedDict) instead of active objects to keep active session states serializable.
  * Captures an immutable 'config_snapshot' of configuration settings at call startup. This ensures that any settings modifications made in the dashboard during an active call do not pollute the current running pipeline execution.
  * Exposes helper methods to write telemetry metrics directly from observer callbacks into the session registry.

import os
from collections.abc import MutableMapping
from typing import Any, TypedDict
from app.config import Settings, V1_SAMPLE_RATE

class ConfigSnapshot(TypedDict):
    """Immutable configuration snapshot stored at call start."""
    SYSTEM_PROMPT: str
    LLM_MODEL: str
    LLM_TEMPERATURE: float
    TTS_VOICE_ID: str
    TTS_MODEL: str
    INITIAL_GREETING_PROMPT: str
    SAMPLE_RATE: int

class RuntimeSession(TypedDict):
    """Serializable runtime session state shared across phases."""
    call_sid: str
    stream_sid: str
    metrics: dict[str, Any]
    transcript: list[dict[str, Any]]
    first_latency: float | None
    config_snapshot: ConfigSnapshot

ACTIVE_SESSIONS: MutableMapping[str, RuntimeSession] = {}

def build_config_snapshot(settings: Settings) -> ConfigSnapshot:
    """Create the fixed v1 runtime configuration snapshot for a call."""
    return {
        "SYSTEM_PROMPT": settings.system_prompt,
        "LLM_MODEL": settings.llm_model,
        "LLM_TEMPERATURE": settings.llm_temperature,
        "TTS_VOICE_ID": settings.tts_voice_id,
        "TTS_MODEL": settings.tts_model,
        "INITIAL_GREETING_PROMPT": settings.initial_greeting_prompt,
        "SAMPLE_RATE": V1_SAMPLE_RATE,
    }

def create_session(call_sid: str, stream_sid: str, settings: Settings) -> RuntimeSession:
    """Create and register a new active runtime session."""
    if call_sid in ACTIVE_SESSIONS:
        raise ValueError(f"Session already exists for call_sid={call_sid}")
    session: RuntimeSession = {
        "call_sid": call_sid,
        "stream_sid": stream_sid,
        "metrics": {},
        "transcript": [],
        "first_latency": None,
        "config_snapshot": build_config_snapshot(settings),
    }
    ACTIVE_SESSIONS[call_sid] = session
    return session

def get_session(call_sid: str) -> RuntimeSession | None:
    """Return an active runtime session by call SID, if present."""
    return ACTIVE_SESSIONS.get(call_sid)

def delete_session(call_sid: str) -> RuntimeSession | None:
    """Remove and return an active runtime session by call SID, if present."""
    return ACTIVE_SESSIONS.pop(call_sid, None)

def update_session_first_latency(call_sid: str, latency_secs: float) -> None:
    """Set the first-bot-speech latency for a session (once only)."""
    session = ACTIVE_SESSIONS.get(call_sid)
    if session is None:
        return
    if session["first_latency"] is None:
        session["first_latency"] = latency_secs
        session["metrics"]["first_bot_speech_latency_secs"] = latency_secs

def append_session_turn_metrics(call_sid: str, turn_metrics: dict[str, Any]) -> None:
    """Append a completed turn's metrics to the session."""
    session = ACTIVE_SESSIONS.get(call_sid)
    if session is None:
        return
    turns: list[dict[str, Any]] = session["metrics"].setdefault("turns", [])
    turns.append(turn_metrics)
    session["metrics"]["turn_count"] = len(turns)

from fastapi import WebSocket
from pipecat.serializers.exotel import ExotelFrameSerializer
from pipecat.transports.websocket.fastapi import (
    FastAPIWebsocketParams,
    FastAPIWebsocketTransport,
)
from app.config import V1_SAMPLE_RATE

def build_exotel_serializer(stream_sid: str, call_sid: str) -> ExotelFrameSerializer:
    """Build the fixed Exotel serializer for the v1 transport contract."""
    return ExotelFrameSerializer(
        stream_sid=stream_sid,
        call_sid=call_sid,
        params=ExotelFrameSerializer.InputParams(
            exotel_sample_rate=V1_SAMPLE_RATE,
            sample_rate=V1_SAMPLE_RATE,
        ),
    )

def build_exotel_transport(
    websocket: WebSocket,
    stream_sid: str,
    call_sid: str,
) -> FastAPIWebsocketTransport:
    """Build the FastAPI WebSocket transport for an Exotel call."""
    serializer = build_exotel_serializer(stream_sid=stream_sid, call_sid=call_sid)
    return FastAPIWebsocketTransport(
        websocket=websocket,
        params=FastAPIWebsocketParams(
            serializer=serializer,
            audio_in_enabled=True,
            audio_out_enabled=True,
            audio_in_sample_rate=V1_SAMPLE_RATE,
            audio_out_sample_rate=V1_SAMPLE_RATE,
        ),
    )

FILE 3: app/services/pipecat_pipeline.py
================================================================================
- Context: Assembles the STT -> Context Aggregator -> LLM -> TTS pipeline and handles execution.
- Key Design Decisions:
  * Restricts API parameter instantiation to typed Service settings (e.g. CerebrasLLMSettings, CartesiaTTSSettings) in compliance with Pipecat 1.2.1.
  * Uses `LLMContextAggregatorPair` to coordinate dialog turns.
  * Enforces user audio muting when the bot speaks: combines MuteUntilFirstBotCompleteUserMuteStrategy and AlwaysUserMuteStrategy. This ensures a non-interruptible dialog.
  * Keeps SileroVADAnalyzer inside the context aggregator pair to support the latency metrics observers which rely on timing packets.
  * Pre-queues the bot greeting in the Task queue using LLMMessagesAppendFrame. This lets the bot speak first immediately after connection, without requiring dynamic polling hooks.
  * Returns a serialized `PipelineExecutionResult` wrapper to prevent live Pipecat pipeline objects from leaking back to HTTP controllers.

from __future__ import annotations
import asyncio
from typing import Any
from loguru import logger
from pipecat.audio.vad.silero import SileroVADAnalyzer
from pipecat.frames.frames import LLMMessagesAppendFrame
from pipecat.observers.user_bot_latency_observer import UserBotLatencyObserver
from pipecat.pipeline.pipeline import Pipeline
from pipecat.pipeline.runner import PipelineRunner
from pipecat.pipeline.task import PipelineParams, PipelineTask
from pipecat.processors.aggregators.llm_context import LLMContext
from pipecat.processors.aggregators.llm_response_universal import (
    LLMContextAggregatorPair,
    LLMUserAggregatorParams,
)
from pipecat.services.cartesia.tts import CartesiaTTSService, CartesiaTTSSettings
from pipecat.services.cerebras.llm import CerebrasLLMService, CerebrasLLMSettings
from pipecat.services.deepgram.stt import DeepgramSTTService, DeepgramSTTSettings
from pipecat.services.tts_service import TextAggregationMode
from pipecat.transports.websocket.fastapi import FastAPIWebsocketTransport
from pipecat.turns.user_mute import (
    AlwaysUserMuteStrategy,
    MuteUntilFirstBotCompleteUserMuteStrategy,
)

from app.config import Settings, V1_SAMPLE_RATE
from app.runtime import ConfigSnapshot, append_session_turn_metrics, update_session_first_latency
from app.utils.metrics_logger import build_turn_metrics

def _build_stt(settings: Settings) -> DeepgramSTTService:
    """Construct the Deepgram STT service with Flux model.
    Verified Pipecat 1.2.1 pattern:
    - sample_rate passed as direct constructor kwarg
    - STT options via settings=DeepgramSTTSettings(...)
    """
    return DeepgramSTTService(
        api_key=settings.deepgram_api_key,
        sample_rate=V1_SAMPLE_RATE,
        settings=DeepgramSTTSettings(
            model="flux",
            language="en",
            interim_results=False,
        ),
    )

def _build_llm(snapshot: ConfigSnapshot, settings: Settings) -> CerebrasLLMService:
    """Construct the Cerebras LLM service using typed Settings."""
    return CerebrasLLMService(
        api_key=settings.cerebras_api_key,
        settings=CerebrasLLMSettings(
            model=snapshot["LLM_MODEL"],
            temperature=snapshot["LLM_TEMPERATURE"],
        ),
    )

def _build_tts(snapshot: ConfigSnapshot, settings: Settings) -> CartesiaTTSService:
    """Construct the Cartesia TTS service with SENTENCE aggregation."""
    return CartesiaTTSService(
        api_key=settings.cartesia_api_key,
        sample_rate=V1_SAMPLE_RATE,
        text_aggregation_mode=TextAggregationMode.SENTENCE,
        max_buffer_delay_ms=0,
        settings=CartesiaTTSSettings(
            voice=snapshot["TTS_VOICE_ID"],
            model=snapshot["TTS_MODEL"],
        ),
    )

def _build_context(snapshot: ConfigSnapshot) -> LLMContext:
    """Build the initial LLM context with the system prompt."""
    return LLMContext(
        messages=[
            {"role": "system", "content": snapshot["SYSTEM_PROMPT"]},
        ],
    )

def _build_aggregator_pair(context: LLMContext) -> LLMContextAggregatorPair:
    """Build the user/bot aggregator pair with muted turn management."""
    user_params = LLMUserAggregatorParams(
        vad_analyzer=SileroVADAnalyzer(
            sample_rate=V1_SAMPLE_RATE,
        ),
        user_mute_strategies=[
            MuteUntilFirstBotCompleteUserMuteStrategy(),
            AlwaysUserMuteStrategy(),
        ],
    )
    return LLMContextAggregatorPair(
        context=context,
        user_params=user_params,
    )

class PipelineExecutionResult:
    """Serializable result returned after pipeline execution completes."""
    def __init__(self, *, completed: bool, error: str | None = None) -> None:
        self.completed = completed
        self.error = error

async def run_pipeline(
    *,
    transport: FastAPIWebsocketTransport,
    call_sid: str,
    stream_sid: str,
    snapshot: ConfigSnapshot,
    settings: Settings,
) -> PipelineExecutionResult:
    """Assemble and run the Pipecat pipeline for one call."""
    log = logger.bind(call_sid=call_sid)

    # Service construction
    stt = _build_stt(settings)
    llm = _build_llm(snapshot, settings)
    tts = _build_tts(snapshot, settings)

    context = _build_context(snapshot)
    aggregator_pair = _build_aggregator_pair(context)

    # Pipeline assembly
    pipeline = Pipeline(
        [
            transport.input(),
            stt,
            aggregator_pair.user(),
            llm,
            tts,
            aggregator_pair.assistant(),
            transport.output(),
        ]
    )

    # Observer registration on PipelineTask (not PipelineParams)
    latency_observer = UserBotLatencyObserver()

    task = PipelineTask(
        pipeline,
        params=PipelineParams(
            enable_metrics=True,
            audio_in_sample_rate=V1_SAMPLE_RATE,
            audio_out_sample_rate=V1_SAMPLE_RATE,
        ),
        observers=[latency_observer],
    )

    _pending_latency: dict[str, float] = {}

    @latency_observer.event_handler("on_latency_measured")
    async def _on_latency_measured(observer: Any, latency_seconds: float) -> None:
        log.info("Turn latency measured", latency_secs=f"{latency_seconds:.3f}")
        _pending_latency["value"] = latency_seconds

    @latency_observer.event_handler("on_latency_breakdown")
    async def _on_latency_breakdown(observer: Any, breakdown: Any) -> None:
        latency_secs = _pending_latency.pop("value", None)
        turn = build_turn_metrics(latency_secs=latency_secs, breakdown=breakdown)
        append_session_turn_metrics(call_sid, turn.model_dump())
        log.info("Turn breakdown captured", events_count=len(turn.latency_events))

    @latency_observer.event_handler("on_first_bot_speech_latency")
    async def _on_first_bot_speech(observer: Any, latency_seconds: float) -> None:
        log.info("First bot speech latency", latency_secs=f"{latency_seconds:.3f}")
        update_session_first_latency(call_sid, latency_seconds)

    @transport.event_handler("on_client_disconnected")
    async def _on_client_disconnected(transport_ref: Any, client: Any) -> None:
        log.info("Transport client disconnected, cancelling pipeline task")
        await task.cancel()

    greeting_frame = LLMMessagesAppendFrame(
        messages=[
            {"role": "user", "content": snapshot["INITIAL_GREETING_PROMPT"]},
        ],
    )
    await task.queue_frame(greeting_frame)

    runner = PipelineRunner(handle_sigint=False)
    log.info("Pipeline runner starting", stream_sid=stream_sid)

    try:
        await runner.run(task)
        log.info("Pipeline runner completed normally")
        return PipelineExecutionResult(completed=True)
    except asyncio.CancelledError:
        log.info("Pipeline runner cancelled (user disconnected)")
        return PipelineExecutionResult(completed=True, error="User disconnected")
    except Exception as exc:
        log.error("Pipeline runner failed: {error}", error=str(exc))
        return PipelineExecutionResult(completed=False, error=str(exc))

FILE 4: app/routers/telephony.py
================================================================================
- Context: FastAPI controller managing call webhooks and the websocket lifecycle.
- Key Design Decisions:
  * Separates the Exotel answer webhook (`GET /telephony/webhook`, which resolves dynamic ngrok paths) from the WebSocket streaming endpoint (`WS /telephony/exotel`).
  * Integrates a robust `finally` safety net. Whether the call completes normally, disconnects early, or triggers an unexpected exception, it will always:
    1. Parse the metrics.
    2. Persist the call record into MongoDB.
    3. Clear the active session registry.
  * Isolates the database call inside a `try/except` block to prevent persistence failures from masking connection teardown logs.

from collections.abc import Mapping
from typing import Any
from fastapi import APIRouter, Request, WebSocket, WebSocketDisconnect
from fastapi.responses import JSONResponse
from loguru import logger
from pipecat.runner.utils import parse_telephony_websocket

from app.config import get_settings
from app.database import persist_call_log
from app.runtime import create_session, delete_session, get_session
from app.services.pipecat_pipeline import run_pipeline
from app.services.transport_factory import build_exotel_transport
from app.utils.metrics_logger import finalize_call_log

POLICY_VIOLATION_CODE = 1008
UNSUPPORTED_PROVIDER_REASON = "unsupported_provider"
MISSING_IDENTITY_REASON = "missing_exotel_identity"
DUPLICATE_CALL_SID_REASON = "duplicate_call_sid"

router = APIRouter(prefix="/telephony", tags=["telephony"])

@router.get("/webhook")
async def exotel_webhook(request: Request) -> JSONResponse:
    """Handle Exotel Voicebot Applet dynamic URL callback."""
    settings = get_settings()
    call_sid = request.query_params.get("CallSid", "unknown")
    log = logger.bind(call_sid=call_sid)
    log.info(
        "Exotel webhook received",
        call_from=request.query_params.get("CallFrom"),
        call_to=request.query_params.get("CallTo"),
        direction=request.query_params.get("Direction"),
    )
    clean_ngrok = settings.public_ngrok_url.replace("https://", "").replace("http://", "").strip("/")
    wss_url = f"wss://{clean_ngrok}/telephony/exotel?sample-rate=16000"
    log.info("Returning WebSocket URL to Exotel", wss_url=wss_url)
    return JSONResponse(content={"url": wss_url})

def _resolve_query_identity(websocket: WebSocket) -> tuple[str | None, str | None]:
    """Read Exotel identity values from the initial query string."""
    return (
        websocket.query_params.get("CallSid"),
        websocket.query_params.get("StreamSid"),
    )

def _remap_exotel_identity(call_data: Mapping[str, Any]) -> tuple[str | None, str | None]:
    """Normalize Pipecat Exotel call data into repo-standard naming."""
    call_sid = call_data.get("call_id")
    stream_sid = call_data.get("stream_id")
    if isinstance(call_sid, str) and isinstance(stream_sid, str):
        return call_sid, stream_sid
    return None, None

@router.websocket("/exotel")
async def exotel_websocket(websocket: WebSocket) -> None:
    """Handle Exotel WebSocket transport bootstrap and pipeline execution."""
    created_call_sid: str | None = None
    await websocket.accept()

    try:
        call_sid, stream_sid = _resolve_query_identity(websocket)
        if not call_sid or not stream_sid:
            try:
                transport_type, call_data = await parse_telephony_websocket(websocket)
            except ValueError:
                await websocket.close(code=POLICY_VIOLATION_CODE, reason=MISSING_IDENTITY_REASON)
                return
            if transport_type != "exotel":
                await websocket.close(code=POLICY_VIOLATION_CODE, reason=UNSUPPORTED_PROVIDER_REASON)
                return
            call_sid, stream_sid = _remap_exotel_identity(call_data)

        if not call_sid or not stream_sid:
            await websocket.close(code=POLICY_VIOLATION_CODE, reason=MISSING_IDENTITY_REASON)
            return

        settings = get_settings()
        try:
            session = create_session(call_sid=call_sid, stream_sid=stream_sid, settings=settings)
        except ValueError:
            await websocket.close(code=POLICY_VIOLATION_CODE, reason=DUPLICATE_CALL_SID_REASON)
            return

        created_call_sid = call_sid
        log = logger.bind(call_sid=call_sid, stream_sid=stream_sid)
        log.info("Session created, building transport")

        transport = build_exotel_transport(websocket=websocket, stream_sid=stream_sid, call_sid=call_sid)
        log.info("Transport ready, handing off to pipeline service")

        result = await run_pipeline(
            transport=transport,
            call_sid=call_sid,
            stream_sid=stream_sid,
            snapshot=session["config_snapshot"],
            settings=settings,
        )

        if result.error:
            log.warning("Pipeline completed with error: {error}", error=result.error)
        else:
            log.info("Pipeline completed normally")

        session = get_session(call_sid)
        if session is not None:
            call_log = finalize_call_log(
                call_sid=call_sid,
                stream_sid=stream_sid,
                config_snapshot=session["config_snapshot"],
                metrics=session["metrics"],
                completed=result.completed,
                error=result.error,
            )
            try:
                doc_id = await persist_call_log(call_log)
                log.info("Call log persisted", document_id=doc_id)
            except Exception:
                log.exception("Failed to persist call log to MongoDB")

    except WebSocketDisconnect:
        if created_call_sid:
            log = logger.bind(call_sid=created_call_sid)
            log.info("WebSocket disconnected before pipeline execution started")
            session = get_session(created_call_sid)
            if session is not None:
                call_log = finalize_call_log(
                    call_sid=created_call_sid,
                    stream_sid=session["stream_sid"],
                    config_snapshot=session["config_snapshot"],
                    metrics=session["metrics"],
                    completed=False,
                    error="WebSocket disconnected before pipeline execution",
                )
                try:
                    doc_id = await persist_call_log(call_log)
                    log.info("Call log persisted after pre-pipeline disconnect", document_id=doc_id)
                except Exception:
                    log.exception("Failed to persist call log to MongoDB")
    except Exception:
        if created_call_sid:
            log = logger.bind(call_sid=created_call_sid)
            log.exception("Unexpected error in telephony route")
            session = get_session(created_call_sid)
            if session is not None:
                call_log = finalize_call_log(
                    call_sid=created_call_sid,
                    stream_sid=session["stream_sid"],
                    config_snapshot=session["config_snapshot"],
                    metrics=session["metrics"],
                    completed=False,
                    error="Unexpected error during call handling",
                )
                try:
                    doc_id = await persist_call_log(call_log)
                    log.info("Call log persisted after unexpected error", document_id=doc_id)
                except Exception:
                    log.exception("Failed to persist call log to MongoDB")
        raise
    finally:
        if created_call_sid is not None:
            delete_session(created_call_sid)
            logger.bind(call_sid=created_call_sid).info("Session cleaned up from ACTIVE_SESSIONS")

FILE 5: app/utils/metrics_logger.py
================================================================================
- Context: Normalizes raw pipeline metrics and packages data for database storage.
- Key Design Decisions:
  * Pipecat's internal `chronological_events()` method outputs string descriptions that throw away start_time floats. We manually access the raw `LatencyBreakdown` attributes directly (`.ttfb`, `.text_aggregation`, `.user_turn_start_time`, `.user_turn_secs`, `.function_calls`) to keep exact timestamp values.
  * Accesses these attributes using duck-typing, avoiding the need to import heavy Pipecat libraries into our lightweight schema normalization code.
  * Groups values into Pydantic models (e.g. `TurnMetrics`, `CallMetrics`, `CallLog`) to guarantee strict datatype validation before inserting data into MongoDB.

--- FILE CONTENT: app/utils/metrics_logger.py ---
from __future__ import annotations
from typing import Any
from loguru import logger
from app.models.schemas import CallLog, CallMetrics, LatencyEvent, TurnMetrics

def normalize_latency_breakdown(breakdown: Any) -> list[dict[str, Any]]:
    """Normalize a LatencyBreakdown into persistence-safe dictionaries."""
    events: list[tuple[float, str]] = []

    if breakdown.user_turn_start_time is not None and breakdown.user_turn_secs is not None:
        events.append((
            breakdown.user_turn_start_time,
            f"User turn: {breakdown.user_turn_secs:.3f}s",
        ))

    for t in breakdown.ttfb:
        events.append((t.start_time, f"{t.processor}: TTFB {t.duration_secs:.3f}s"))

    for fc in breakdown.function_calls:
        events.append((fc.start_time, f"{fc.function_name}: {fc.duration_secs:.3f}s"))

    if breakdown.text_aggregation:
        ta = breakdown.text_aggregation
        events.append((
            ta.start_time,
            f"{ta.processor}: text aggregation {ta.duration_secs:.3f}s",
        ))

    events.sort(key=lambda e: e[0])
    return [{"time": t, "event": ev} for t, ev in events]

def build_turn_metrics(*, latency_secs: float | None, breakdown: Any | None) -> TurnMetrics:
    """Build a TurnMetrics from a single user-to-bot turn."""
    latency_events: list[LatencyEvent] = []
    if breakdown is not None:
        raw = normalize_latency_breakdown(breakdown)
        latency_events = [LatencyEvent(time=e["time"], event=e["event"]) for e in raw]

    return TurnMetrics(
        user_bot_latency_secs=latency_secs,
        latency_events=latency_events,
    )

def finalize_call_log(
    *,
    call_sid: str,
    stream_sid: str,
    config_snapshot: dict[str, Any],
    metrics: dict[str, Any],
    completed: bool,
    error: str | None,
) -> CallLog:
    """Build a persistence-safe CallLog from runtime session state."""
    log = logger.bind(call_sid=call_sid)

    call_metrics = CallMetrics(
        first_bot_speech_latency_secs=metrics.get("first_bot_speech_latency_secs"),
        turn_count=metrics.get("turn_count", 0),
        turns=metrics.get("turns", []),
    )

    call_log = CallLog(
        call_sid=call_sid,
        stream_sid=stream_sid,
        config_snapshot=config_snapshot,
        metrics=call_metrics,
        completed=completed,
        error=error,
    )

    log.info(
        "Call log finalized",
        turn_count=call_metrics.turn_count,
        first_latency=call_metrics.first_bot_speech_latency_secs,
    )
    return call_log
